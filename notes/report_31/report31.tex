\title{Twin-space system Liouvillian}
Consider the Hamiltonian 
\begin{equation}
	H = V(d_1^\dagger d_2^\pdagger + d_2^\dagger d_1^\pdagger).
\end{equation}
The Liouville operator is
\begin{align}
	\mathcal{L} \rho &\equiv [H,\rho]\\
	&=V(d_1^\dagger d_2^\pdagger\rho + d_2^\dagger d_1^\pdagger \rho - \rho d_1^\dagger d_2^\pdagger - \rho d_2^\dagger d_1^\pdagger)\\
	\rightarrow \mathcal{L}\vert \rho\rangle\!\rangle &= V(\hat{d}_1^\dagger \hat{d}_2^\pdagger + \hat{d}_2^\dagger \hat{d}_1^\pdagger - \tilde{d}_2^\dagger \tilde{d}_1^\pdagger - \tilde{d}_1^\dagger \tilde{d}_2^\pdagger)\vert \rho\rangle\!\rangle \\
	&\equiv (\hat{H}-\tilde{H})\vert \rho\rangle\!\rangle\label{eq:equation_5}
\end{align}
where the last step is obtained using
\begin{align*}
	d^\pm \rho &\equiv \hat{d}^\pm \vert \rho\rangle\!\rangle\\
	\rho  d^\pm &\equiv \tilde{d}^\mp \vert \rho\rangle\!\rangle
\end{align*}
and consequently one has
\begin{align}
	\rho d_1^\dagger d_2^\pdagger &= (\rho d_1^\dagger) d_2^\pdagger \\
	&= \tilde{d}_2^\dagger \tilde{d}_1^\pdagger \vert \rho\rangle\!\rangle.
\end{align}
In \cref{eq:equation_5} we have defined
\begin{align*}
	\hat{H}&=\hat{d}_1^\dagger \hat{d}_2^\pdagger + \hat{d}_2^\dagger \hat{d}_1^\pdagger\\
	\tilde{H}&=\tilde{d}_2^\dagger \tilde{d}_1^\pdagger + \tilde{d}_1^\dagger \tilde{d}_2^\pdagger.
\end{align*}
